import os
import sys
import argparse
import io
from pypdf import PdfReader, PdfWriter
from PIL import Image

def get_file_size_mb(path: str) -> float:
    return os.path.getsize(path) / (1024 * 1024)

def format_size(mb: float) -> str:
    if mb < 1.0:
        return f"{mb * 1024:.1f} KB"
    return f"{mb:.2f} MB"

def compress_pdf(input_path: str, output_path: str, level: int):
    reader = PdfReader(input_path)
    writer = PdfWriter()

    img_settings = {
        1: None,
        2: {"quality": 75, "max_dim": 1600},
        3: {"quality": 40, "max_dim": 1000},
    }

    selected_setting = img_settings[level]

    for page in reader.pages:
        if selected_setting is not None:
            for img_name, img_obj in page.images.items():
                try:
                    raw_img = Image.open(io.BytesIO(img_obj.data))
                    if raw_img.mode in ("RGBA", "P"):
                        raw_img = raw_img.convert("RGB")

                    w, h = raw_img.size
                    max_dim = selected_setting["max_dim"]
                    if max(w, h) > max_dim:
                        scaling = max_dim / float(max(w, h))
                        raw_img = raw_img.resize(
                            (int(w * scaling), int(h * scaling)),
                            Image.Resampling.LANCZOS
                        )

                    compressed_io = io.BytesIO()
                    raw_img.save(
                        compressed_io,
                        format="JPEG",
                        quality=selected_setting["quality"],
                        optimize=True
                    )
                    compressed_io.seek(0)
                    page.images[img_name].replace(compressed_io.getvalue())
                except Exception:
                    continue

        page.compress_content_streams()
        writer.add_page(page)

    writer.add_metadata({})

    with open(output_path, "wb") as f_out:
        writer.write(f_out)

def interactive_mode():
    print("=" * 55)
    print("        PDF COMPRESSION UTILITY (INTERACTIVE)       ")
    print("=" * 55)

    input_file = input("Enter path to PDF file: ").strip().strip('"').strip("'")
    if not os.path.exists(input_file) or not input_file.lower().endswith(".pdf"):
        print(f"Error: Invalid PDF file path: {input_file}")
        sys.exit(1)

    orig_size = get_file_size_mb(input_file)
    print(f"\nOriginal Size: {format_size(orig_size)}")
    print("-" * 55)
    print(" [1] Light (~10-20% reduction, lossless stream compression)")
    print(" [2] Medium (~40-60% reduction, 150 DPI, 75% quality)")
    print(" [3] Aggressive (~70-85% reduction, 96 DPI, 40% quality)")
    print("-" * 55)

    choice = input("Select level (1, 2, or 3): ").strip()
    if choice not in ("1", "2", "3"):
        print("Invalid choice.")
        sys.exit(1)

    level = int(choice)
    base, ext = os.path.splitext(input_file)
    default_out = f"{base}_compressed_level{level}{ext}"
    out_prompt = input(f"Output path [Enter for '{default_out}']: ").strip()
    output_file = out_prompt if out_prompt else default_out

    return input_file, output_file, level

def main():
    parser = argparse.ArgumentParser(
        description="Multi-tier PDF Compression CLI Tool",
        epilog="Run without arguments to launch interactive wizard."
    )
    parser.add_argument("input", nargs="?", help="Path to input PDF file")
    parser.add_argument("-l", "--level", type=int, choices=[1, 2, 3], help="Compression tier (1=Light, 2=Medium, 3=Aggressive)")
    parser.add_argument("-o", "--output", help="Path for compressed output PDF")

    args = parser.parse_args()

    # If no input file is supplied via CLI flags, launch interactive mode
    if args.input is None:
        input_file, output_file, level = interactive_mode()
    else:
        input_file = args.input.strip().strip('"').strip("'")
        if not os.path.exists(input_file) or not input_file.lower().endswith(".pdf"):
            print(f"Error: File not found or not a PDF: {input_file}")
            sys.exit(1)

        level = args.level if args.level is not None else 2  # Default to level 2 if omitted

        if args.output:
            output_file = args.output
        else:
            base, ext = os.path.splitext(input_file)
            output_file = f"{base}_compressed_level{level}{ext}"

    orig_size = get_file_size_mb(input_file)
    print(f"\nCompressing '{input_file}' using Level {level}...")

    compress_pdf(input_file, output_file, level)

    final_size = get_file_size_mb(output_file)
    savings = ((orig_size - final_size) / orig_size) * 100 if orig_size > 0 else 0

    print("\nDone!")
    print(f"Output file:     {output_file}")
    print(f"Original size:   {format_size(orig_size)}")
    print(f"Compressed size: {format_size(final_size)}")
    print(f"Space saved:     {savings:.1f}%\n")

if __name__ == "__main__":
    main()
